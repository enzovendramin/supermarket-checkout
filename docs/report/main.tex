% Helpers that let a chapter share the page with what precedes it, so a part
% divider (or a short appendix) does not waste a whole page.
\makeatletter
\let\spm@chapter\chapter
\let\spm@normalchapterhead\@makechapterhead
% A compact chapter head used only by a chapter that shares a page; it restores
% the normal head automatically after running once.
\newcommand{\spm@tightchapterhead}[1]{%
  \vspace*{6\p@}%
  {\parindent\z@ \raggedright \normalfont
    \ifnum \c@secnumdepth >\m@ne
      \huge\bfseries \@chapapp\space \thechapter \par\nobreak \vskip 8\p@
    \fi
    \interlinepenalty\@M
    \Huge\bfseries #1\par\nobreak \vskip 18\p@}%
  \global\let\@makechapterhead\spm@normalchapterhead}
% The next chapter continues on the current page (compact head), then both the
% page-break behaviour and the head are restored.
\newcommand{\spm@sharepage}{%
  \global\let\chapter\spm@chapter
  \global\let\@makechapterhead\spm@tightchapterhead
  \thispagestyle{plain}%
  \global\@topnum\z@
  \@afterindentfalse
  \secdef\@chapter\@schapter}
\newcommand{\nopagebreakchapter}{\renewcommand{\chapter}{\spm@sharepage}}
\newcommand{\partnopage}[1]{%
  \clearpage
  \thispagestyle{plain}%
  \refstepcounter{part}%
  \addcontentsline{toc}{part}{\thepart\hspace{1em}#1}%
  {\centering\normalfont\Huge\bfseries\partname~\thepart\par
   \vspace{0.4em}#1\par}%
  \vspace{0.4em}%
  \nopagebreakchapter
}
\makeatother
